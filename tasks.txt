task structure stored in the dictionary (e.g. in the data field area of the task dictionary entry)                                                            
------------ -------------- -----------------------------------------
+cell_offset description    value n                                  
                                                                     
0            wake_at        IF ms? >= n THEN semaphore check ELSE next_task                      
 
1            *semaphore     n == 0: no semaphore used -> activate this task               
                            n != 0: IF *n == 0 THEN next_task ELSE activate this task

2            *next_task      addr of the next task, 0 = this is the last task in the list
                                                                     
3            priority       lower means higher priority              
                                                                     
4            DSP            data stack pointer                       
                                                                     
5            RSP            return stack pointer

6            DS bottom (32 cells)

38           RS bottom (16 cells)
------------ -------------- -----------------------------------------

(cell *)first_task //pointer to the top of the sorted task list
(cell *)task //current running task


switch_task ( n -- ) // n = suspend current task for n ms

	// compute new wake_at_ms field for the current task
		ms? // get actuals ms
		task->wake_at = ms? + n
	
	// save context
		PUSHR IP
		task->RSP = RSP
		task->DSP = DSP + 1 // we are not saving TOS as TOS is consumed by switch_task WORD, only DSP is decremented while TOS is consumed 

	// find task to switch to

		start_from_the_first_task:
         ms? // get actuals ms
			task = first_task
			first_wake_at = task->wake_at

		loop_thru_tasks:
			// check wake_at_ms
				IF ms? >= task->wake_at THEN semaphore_check
					ELSE next_task

			// check semaphore
				IF task->semaphore == 0 OR *task->semaphore != 0 THEN load_context
					ELSE next_task

			next_task:
				// find the task which needs to be activated in the shortert time
				IF task->wake_at < first_wake_at THEN first_wake_at = task->wake_at

				// is this the last task in list? if so, we can suspend the FORTH system
					IF task->next_task == 0 THEN
						sleep(first_wake_at - ms?)
						start_from_the_first_task
					ELSE
						task = task->next_task
						loop_thru_tasks
					
      load_context:
			
			DSP = task->DSP
			POP TOS
			
			RSP = task->RSP
			POPR IP

			NEXT


add_task ( n -- ) // add new task to scheduler's list, n is a address of task structure. Task is inserted to the task list sorted by priorities (lowest first)
   POP (cell *)new_task
   IF new_task->priority <= first_task->priority THEN
      new_task->next_task = *first_task
      first_task = new_task
   
   ELSE
      t = (cell *)first_task
      WHILE (t->next_task != 0) AND (t->next_task->priority < new_task->priority)
            t = t->next

      new_task->next_task = t->next_task
      t->next_task = new_task
   NEXT
